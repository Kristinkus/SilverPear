package com.example.silverpear.service;

import com.example.silverpear.concurrency.dto.AsyncTaskState;
import com.example.silverpear.concurrency.dto.AsyncTaskStatusResponse;
import com.example.silverpear.concurrency.dto.CounterSnapshotResponse;
import com.example.silverpear.concurrency.dto.RaceConditionResultResponse;
import lombok.extern.slf4j.Slf4j;
import org.springframework.scheduling.annotation.Async;
import org.springframework.stereotype.Service;

import java.util.UUID;
import java.util.concurrent.CompletableFuture;
import java.util.concurrent.ConcurrentHashMap;
import java.util.concurrent.CountDownLatch;
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

@Slf4j
@Service
public class ConcurrencyDemoService {

    private final ConcurrentHashMap<String, AsyncTaskStatusResponse> taskStatuses = new ConcurrentHashMap<>();
    private final AtomicInteger atomicCounter = new AtomicInteger(0);
    private int synchronizedCounter = 0;
    private int unsafeCounter = 0;

    public String startAsyncBusinessOperation(String payload) {
        String taskId = UUID.randomUUID().toString();
        AsyncTaskStatusResponse status = new AsyncTaskStatusResponse();
        status.setTaskId(taskId);
        status.setState(AsyncTaskState.PENDING);
        status.setProgressPercent(0);
        taskStatuses.put(taskId, status);

        processTaskAsync(taskId, payload);
        return taskId;
    }

    public AsyncTaskStatusResponse getTaskStatus(String taskId) {
        return taskStatuses.get(taskId);
    }

    @Async
    public CompletableFuture<Void> processTaskAsync(String taskId, String payload) {
        AsyncTaskStatusResponse status = taskStatuses.get(taskId);
        if (status == null) {
            return CompletableFuture.completedFuture(null);
        }

        try {
            status.setState(AsyncTaskState.RUNNING);
            for (int progress = 20; progress <= 100; progress += 20) {
                TimeUnit.MILLISECONDS.sleep(350);
                status.setProgressPercent(progress);
            }
            status.setResult("Задача выполнена успешно. payload=" + payload);
            status.setState(AsyncTaskState.COMPLETED);
            log.info("Async task completed: {}", taskId);
        } catch (InterruptedException ex) {
            Thread.currentThread().interrupt();
            status.setState(AsyncTaskState.FAILED);
            status.setError("Task interrupted");
            log.warn("Async task interrupted: {}", taskId, ex);
        } catch (Exception ex) {
            status.setState(AsyncTaskState.FAILED);
            status.setError(ex.getMessage());
            log.error("Async task failed: {}", taskId, ex);
        }
        return CompletableFuture.completedFuture(null);
    }

    public CounterSnapshotResponse incrementThreadSafeCounters(int times) {
        for (int i = 0; i < times; i++) {
            incrementSynchronizedCounter();
            atomicCounter.incrementAndGet();
        }
        return getCounterSnapshot();
    }

    public CounterSnapshotResponse getCounterSnapshot() {
        return new CounterSnapshotResponse(getSynchronizedCounter(), atomicCounter.get());
    }

    public void resetCounters() {
        synchronized (this) {
            synchronizedCounter = 0;
            unsafeCounter = 0;
        }
        atomicCounter.set(0);
    }

    public RaceConditionResultResponse runRaceConditionDemo(int threads, int incrementsPerThread) {
        resetCounters();

        int expected = threads * incrementsPerThread;
        long startedAt = System.nanoTime();
        ExecutorService executor = Executors.newFixedThreadPool(threads);
        CountDownLatch startGate = new CountDownLatch(1);
        CountDownLatch finishGate = new CountDownLatch(threads);

        for (int t = 0; t < threads; t++) {
            executor.submit(() -> {
                try {
                    startGate.await();
                    for (int i = 0; i < incrementsPerThread; i++) {
                        unsafeCounter++;
                        incrementSynchronizedCounter();
                        atomicCounter.incrementAndGet();
                    }
                } catch (InterruptedException ex) {
                    Thread.currentThread().interrupt();
                } finally {
                    finishGate.countDown();
                }
            });
        }

        startGate.countDown();
        try {
            finishGate.await(30, TimeUnit.SECONDS);
        } catch (InterruptedException ex) {
            Thread.currentThread().interrupt();
        } finally {
            executor.shutdown();
        }

        long durationMs = TimeUnit.NANOSECONDS.toMillis(System.nanoTime() - startedAt);
        int unsafeActual = unsafeCounter;
        int synchronizedActual = getSynchronizedCounter();
        int atomicActual = atomicCounter.get();

        return new RaceConditionResultResponse(
                threads,
                incrementsPerThread,
                expected,
                unsafeActual,
                synchronizedActual,
                atomicActual,
                expected - unsafeActual,
                durationMs
        );
    }

    private synchronized void incrementSynchronizedCounter() {
        synchronizedCounter++;
    }

    private synchronized int getSynchronizedCounter() {
        return synchronizedCounter;
    }
}
